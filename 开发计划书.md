## 一、架构分层与依赖方向

```
                        ┌─────────────────────────────────────┐
                        │           入口层 (Entry Points)      │
                        │                                     │
                        │   ┌─────┐   ┌─────┐   ┌─────────┐  │
                        │   │ CLI │   │ API │   │MCP Server│  │
                        │   └──┬──┘   └──┬──┘   └────┬─────┘  │
                        │      │        │           │         │
                        └──────┼────────┼───────────┼─────────┘
                               │        │           │
                          ┌────▼────────▼───────────▼────┐
                          │       引擎层 (Engines)        │
                          │                               │
                          │  ┌────────┐ ┌─────┐ ┌──────┐ │
                          │  │ Memory │ │ RAG │ │Prompt│ │
                          │  │ Engine │ │Engine│ │Engine│ │
                          │  └───┬────┘ └──┬──┘ └──┬───┘ │
                          │      │         │       │     │
                          │      └────┬────┘───────┘     │
                          │           │                   │
                          │     ┌─────▼─────┐            │
                          │     │  Search   │            │
                          │     │  Service  │            │
                          │     └─────┬─────┘            │
                          └───────────┼──────────────────┘
                                      │
                          ┌───────────▼──────────────────┐
                          │       存储层 (Storage)        │
                          │                               │
                          │  ┌──────────┐  ┌───────────┐ │
                          │  │  SQLite  │  │ ChromaDB  │ │
                          │  │  Store   │  │  Store    │ │
                          │  └──────────┘  └───────────┘ │
                          └───────────────────────────────┘
                                      │
                          ┌───────────▼──────────────────┐
                          │       模型层 (Models)         │
                          │   Pydantic 数据模型，零依赖     │
                          └──────────────────────────────┘

依赖方向（严格单向）：
  入口层 ──依赖──▶ 引擎层接口（Protocol）
  引擎层 ──依赖──▶ 存储层接口（Protocol）
  所有层 ──依赖──▶ 模型层（纯数据定义）

禁止：
  ✗ 入口层直接访问存储层
  ✗ 存储层反向依赖引擎层
  ✗ 引擎层之间互相依赖（通过 SearchService 聚合）
```

```python
# src/memora/memory/engine.py — 实现
from memora.storage import Storage
from memora.models.memory import Memory, MemoryType, ConversationSummary


class MemoryEngineImpl:
    """MemoryEngine 的具体实现。"""

    def __init__(self, storage: Storage, llm_client: "LLMClient") -> None:
        self._storage = storage
        self._llm = llm_client

    async def extract_from_messages(self, messages: list[dict], source: str = "claude_code") -> list[Memory]:
        # 1. 调用 LLM 结构化提取
        raw_memories = await self._llm.extract_memories(messages)
        # 2. 去重检查（向量相似度 > 0.9 跳过）
        new_memories = await self._deduplicate(raw_memories)
        # 3. 批量写入 SQLite + ChromaDB
        for m in new_memories:
            await self.save(m)
        return new_memories

    async def save(self, memory: Memory) -> Memory:
        # INSERT INTO memories → SQLite
        await self._storage.db.execute(
            "INSERT INTO memories (id, content, memory_type, source, confidence, tags, created_at, updated_at) "
            "VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
            (memory.id, memory.content, memory.memory_type.value, memory.source,
             memory.confidence, ",".join(memory.tags), memory.created_at.isoformat(), memory.updated_at.isoformat()),
        )
        # 向量化 → ChromaDB
        embedding = await self._llm.embed(memory.content)
        await self._storage.vector.upsert("memories", [
            VectorRecord(id=memory.id, embedding=embedding, metadata={"type": memory.memory_type.value})
        ])
        return memory

    async def recall(self, query: str, top_k: int = 5, memory_type: MemoryType | None = None) -> list[Memory]:
        # 1. query → embedding
        query_embedding = await self._llm.embed(query)
        # 2. ChromaDB 向量搜索
        where = {"type": memory_type.value} if memory_type else None
        results = await self._storage.vector.search("memories", query_embedding, top_k=top_k, where=where)
        # 3. 从 SQLite 拉取完整记录
        memories = []
        for r in results:
            row = await self._storage.db.fetch_one("SELECT * FROM memories WHERE id = ?", (r.id,))
            if row:
                memories.append(self._row_to_memory(row))
        return memories

    # ... 其余方法实现
```

```python
# src/memora/rag/engine.py — 实现骨架
from memora.storage import Storage
from memora.models.document import Document, DocumentChunk


class RAGEngineImpl:

    def __init__(self, storage: Storage, llm_client: "LLMClient") -> None:
        self._storage = storage
        self._llm = llm_client
        self._loaders = LoaderRegistry()   # 多格式加载器注册表
        self._chunker = Chunker()           # 分块策略

    async def ingest(self, path_or_url: str) -> Document:
        # 1. 选择加载器 → 提取文本
        loader = self._loaders.get_loader(path_or_url)
        raw_content, metadata = await loader.load(path_or_url)
        # 2. 分块
        chunks = self._chunker.split(raw_content, metadata)
        # 3. 嵌入
        embeddings = await self._llm.embed_batch([c.content for c in chunks])
        # 4. 存储
        doc = Document(title=metadata["title"], source_path=path_or_url, ...)
        await self._save_document(doc, chunks, embeddings)
        return doc
```

```python
# src/memora/search/service.py — 实现
import asyncio
from memora.memory import MemoryEngine
from memora.rag import RAGEngine
from memora.prompt import PromptEngine
from memora.models.search import SearchResult


class SearchServiceImpl:

    def __init__(
        self,
        memory_engine: MemoryEngine,
        rag_engine: RAGEngine,
        prompt_engine: PromptEngine,
    ) -> None:
        self._memory = memory_engine
        self._rag = rag_engine
        self._prompt = prompt_engine

    async def search(self, query: str, scope: str = "all", top_k: int = 10) -> list[SearchResult]:
        tasks = []
        if scope in ("all", "memory"):
            tasks.append(self._search_memory(query, top_k))
        if scope in ("all", "knowledge"):
            tasks.append(self._search_knowledge(query, top_k))
        if scope in ("all", "prompt"):
            tasks.append(self._search_prompts(query, top_k))

        results_list = await asyncio.gather(*tasks)
        merged = [r for group in results_list for r in group]
        merged.sort(key=lambda r: r.relevance_score, reverse=True)
        return merged[:top_k]

    async def _search_memory(self, query: str, top_k: int) -> list[SearchResult]:
        memories = await self._memory.recall(query, top_k=top_k)
        return [SearchResult(
            content=m.content, source_type="memory", source_id=m.id,
            relevance_score=0.0,  # 由向量搜索返回的实际分数填充
            metadata={"type": m.memory_type.value, "tags": m.tags},
        ) for m in memories]

    async def _search_knowledge(self, query: str, top_k: int) -> list[SearchResult]:
        chunks = await self._rag.search(query, top_k=top_k)
        return [SearchResult(
            content=c.content, source_type="document", source_id=c.document_id,
            relevance_score=0.0,
            metadata=c.metadata,
```

### 6.1 应用容器（依赖注入）

```python
# src/memora/app.py
from dataclasses import dataclass
from memora.storage import Storage, create_storage
from memora.memory import MemoryEngine
from memora.memory.engine import MemoryEngineImpl
from memora.rag import RAGEngine
from memora.rag.engine import RAGEngineImpl
from memora.prompt import PromptEngine
from memora.prompt.engine import PromptEngineImpl
from memora.search import SearchService
from memora.search.service import SearchServiceImpl
from memora.llm import LLMClient
from memora.config import load_config


@dataclass
class App:
    """应用容器：持有所有引擎实例，入口层通过此对象获取能力。"""
    storage: Storage
    llm: LLMClient
    memory: MemoryEngine
    rag: RAGEngine
    prompt: PromptEngine
    search: SearchService

    async def shutdown(self) -> None:
        await self.storage.close()


async def create_app(config_path: str = "config/settings.yaml") -> App:
    """工厂函数：组装整个应用。
    
    依赖注入顺序：
    1. 加载配置
    2. 创建 LLM 客户端
    3. 创建存储层
    4. 创建引擎层（注入 storage + llm）
    5. 创建搜索服务（注入三大引擎）
    """

    llm = create_llm_client(config)
    storage = await create_storage(config.db_path, config.chroma_path)

    memory_engine = MemoryEngineImpl(storage, llm)
    rag_engine = RAGEngineImpl(storage, llm)
    prompt_engine = PromptEngineImpl(storage, llm)
    search_service = SearchServiceImpl(memory_engine, rag_engine, prompt_engine)
```

### 6.2 MCP Server

```python
# src/memora/mcp/server.py
from mcp.server import Server
from memora.app import App


def create_mcp_server(app: App) -> Server:
    """创建 MCP Server，注册所有工具。"""
    server = Server("memora")

    @server.tool("save_memory")
    async def save_memory(content: str, memory_type: str, source: str = "claude_code"):
        from memora.models.memory import Memory, MemoryType
        memory = Memory(content=content, memory_type=MemoryType(memory_type), source=source)
        result = await app.memory.save(memory)
        return result.model_dump()

    @server.tool("recall")
    async def recall(query: str, top_k: int = 5, memory_type: str | None = None):
        from memora.models.memory import MemoryType
        mt = MemoryType(memory_type) if memory_type else None
        results = await app.memory.recall(query, top_k=top_k, memory_type=mt)
        return [r.model_dump() for r in results]

    @server.tool("extract_and_save")
    async def extract_and_save(messages: list[dict]):
        results = await app.memory.extract_from_messages(messages)
        return [r.model_dump() for r in results]

    @server.tool("search_knowledge")
    async def search_knowledge(query: str, top_k: int = 5):
        results = await app.rag.search(query, top_k=top_k)
        return [r.model_dump() for r in results]

    @server.tool("ingest_document")
    async def ingest_document(path_or_url: str):
        result = await app.rag.ingest(path_or_url)
        return result.model_dump()

    @server.tool("get_prompt")
    async def get_prompt(name: str):
        prompt, version = await app.prompt.get(name)
        return {"prompt": prompt.model_dump(), "version": version.model_dump()}

    @server.tool("save_prompt")
    async def save_prompt(name: str, content: str, variables: list[str] = []):
        result = await app.prompt.save(name, content, variables=variables)
        return result.model_dump()

    @server.tool("list_memories")
    async def list_memories(memory_type: str | None = None, limit: int = 20):
        from memora.models.memory import MemoryType
        mt = MemoryType(memory_type) if memory_type else None
        results = await app.memory.list(memory_type=mt, limit=limit)
        return [r.model_dump() for r in results]

    @server.tool("get_stats")
    async def get_stats():
        mem_count = len(await app.memory.list(limit=10000))
        doc_count = len(await app.rag.list_documents(limit=10000))
        prompt_count = len(await app.prompt.list())
        return {"memories": mem_count, "documents": doc_count, "prompts": prompt_count}

    return server
```

## 七、核心数据流

### 7.1 MCP 自动记忆提取（核心场景）

```
用户在 Claude Code 中对话
         │
         ▼
Claude Code 识别到有价值信息
         │
         ▼
MCP 调用 extract_and_save(messages)
         │
         ▼
┌─ MemoryEngine.extract_from_messages() ──────────────────┐
│                                                         │
│  1. LLMClient.extract_memories(messages)                │
│     → 结构化提取 [ExtractedMemory, ...]                  │
│                                                         │
│  2. 对每条提取结果：                                      │
│     a. LLMClient.embed(content) → embedding             │
│     b. VectorStore.search("memories", embedding, top_k=1)│
│     c. 若最高相似度 > 0.9 → 跳过（去重）                  │
│     d. 否则 → SQLiteStore.execute(INSERT)               │
│              → VectorStore.upsert("memories", record)   │
│                                                         │
│  3. 返回 [Memory, ...]                                   │
└─────────────────────────────────────────────────────────┘
         │
         ▼
Claude Code 收到提取结果（用户不可见）
```

### 7.2 新对话记忆召回

```
用户在新对话中提问
         │
         ▼
Claude Code 调用 recall("前端技术栈")
         │
         ▼
┌─ MemoryEngine.recall() ─────────────────────────────────┐
│                                                         │
│  1. LLMClient.embed("前端技术栈") → query_embedding      │
│                                                         │
│  2. VectorStore.search("memories", query_embedding)     │
│     → [VectorSearchResult(id, score, metadata), ...]    │
│                                                         │
│  3. 对每个结果：SQLiteStore.fetch_one("SELECT * ...")    │
│     → 组装 Memory 对象                                   │
│                                                         │
│  4. 返回 [Memory, ...]                                   │
└─────────────────────────────────────────────────────────┘
         │
         ▼
Claude Code 将记忆注入上下文，回答时自动引用
```

### 7.3 文档 RAG 检索

```
用户: /search "如何配置认证"
         │
         ▼
SearchService.search("如何配置认证", scope="all")
         │
         ├──→ MemoryEngine.recall("如何配置认证")
         │         → [Memory("上次讨论决定用 JWT", score=0.92)]
         │
         ├──→ RAGEngine.search("如何配置认证")
         │         → [Chunk("Bearer token 认证流程...", score=0.87)]
         │
         └──→ PromptEngine.search("如何配置认证")
                   → []  (无匹配)

         ▼
合并结果，按 relevance_score 降序
         │
         ▼
返回 [SearchResult, SearchResult, ...]
```

## 八、数据库 Schema

### 8.1 SQLite 表结构

```sql
-- memories: 记忆条目
CREATE TABLE memories (
    id          TEXT PRIMARY KEY,
    content     TEXT NOT NULL,
    memory_type TEXT NOT NULL CHECK(memory_type IN ('fact','preference','decision','experience','relationship')),
    source      TEXT DEFAULT '',
    confidence  REAL DEFAULT 1.0,
    tags        TEXT DEFAULT '',          -- 逗号分隔
    access_count INTEGER DEFAULT 0,
    created_at  TEXT NOT NULL,
    updated_at  TEXT NOT NULL
);
CREATE INDEX idx_memories_type ON memories(memory_type);
CREATE INDEX idx_memories_created ON memories(created_at);

-- conversations: 对话摘要
CREATE TABLE conversations (
    id          TEXT PRIMARY KEY,
    title       TEXT NOT NULL,
    summary     TEXT NOT NULL,
    key_points  TEXT DEFAULT '',          -- JSON 数组
    memory_ids  TEXT DEFAULT '',          -- 逗号分隔
    created_at  TEXT NOT NULL
);

-- documents: 文档元数据
CREATE TABLE documents (
    id          TEXT PRIMARY KEY,
    title       TEXT NOT NULL,
    source_path TEXT NOT NULL,
    source_type TEXT NOT NULL CHECK(source_type IN ('file','url','conversation')),
    file_type   TEXT NOT NULL,
    chunk_count INTEGER DEFAULT 0,
    ingested_at TEXT NOT NULL,
    metadata    TEXT DEFAULT '{}'         -- JSON
);

-- document_chunks: 文档分块（元数据，向量在 ChromaDB）
CREATE TABLE document_chunks (
    id          TEXT PRIMARY KEY,
    document_id TEXT NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
    content     TEXT NOT NULL,
    chunk_index INTEGER NOT NULL,
    metadata    TEXT DEFAULT '{}'         -- JSON
);
CREATE INDEX idx_chunks_doc ON document_chunks(document_id);

-- prompts: Prompt 元数据
CREATE TABLE prompts (
    id             TEXT PRIMARY KEY,
    name           TEXT NOT NULL UNIQUE,
    description    TEXT DEFAULT '',
    tags           TEXT DEFAULT '',       -- 逗号分隔
    latest_version INTEGER DEFAULT 0,
    created_at     TEXT NOT NULL
);

-- prompt_versions: Prompt 版本历史
CREATE TABLE prompt_versions (
    id         TEXT PRIMARY KEY,
    prompt_id  TEXT NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
    version    INTEGER NOT NULL,
    content    TEXT NOT NULL,
    variables  TEXT DEFAULT '',           -- JSON 数组
    score      REAL,
    notes      TEXT DEFAULT '',
    created_at TEXT NOT NULL,
    UNIQUE(prompt_id, version)
);
CREATE INDEX idx_pv_prompt ON prompt_versions(prompt_id);

-- tags: 标签
CREATE TABLE tags (
    id   TEXT PRIMARY KEY,
    name TEXT NOT NULL UNIQUE
);

-- memory_tags: 记忆-标签关联
CREATE TABLE memory_tags (
    memory_id TEXT NOT NULL REFERENCES memories(id) ON DELETE CASCADE,
    tag_id    TEXT NOT NULL REFERENCES tags(id) ON DELETE CASCADE,
    PRIMARY KEY (memory_id, tag_id)
);

-- search_history: 搜索历史
CREATE TABLE search_history (
    id         TEXT PRIMARY KEY,
    query      TEXT NOT NULL,
    scope      TEXT NOT NULL,
    result_count INTEGER DEFAULT 0,
    created_at TEXT NOT NULL
);
```

## 十三、关键设计原则

1. **Protocol 隔离**：每个引擎定义 Protocol，实现细节对外不可见。替换实现只需改工厂函数。

2. **单向依赖**：入口层 → 引擎层 → 存储层 → 模型层。禁止反向依赖和跨层调用。

3. **工厂组装**：`create_app()` 是唯一的组装点。所有依赖关系在此处显式声明。

4. **模型零依赖**：models/ 不 import 任何其他模块，是所有模块的公共基础。

5. **存储透明**：引擎层通过 `Storage` 门面访问数据，不直接操作 SQLite/ChromaDB API。

6. **LLM 抽象**：引擎层通过 `LLMClient` Protocol 调用大模型，可随时切换提供商。
